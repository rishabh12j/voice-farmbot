# Runtime dependencies for the growmate_voice package.
#
# These are the NON-ROS2 Python deps. The ROS2 side (rclpy, std_msgs) comes
# from your sourced ROS2 workspace — do not pip-install those.
#
# Install with:
#   pip install -r src/growmate_voice/requirements.txt

# Config parsing (also declared in package.xml as python3-yaml)
pyyaml>=6.0

# Gradio web app — needed for voice_app
gradio>=4.0

# Speech-to-text for the browser mic. faster-whisper is the recommended
# backend; it's CPU-friendly and runs acceptably on a Pi 5. If you can't
# install it for some reason, openai-whisper is a slower fallback the code
# will use automatically.
faster-whisper>=1.0

# Text-to-speech. pyttsx3 uses system voices on macOS / Windows / Linux.
# On the Pi, install `piper` separately and set the PIPER_VOICE env var
# for much better quality; the code will prefer Piper over pyttsx3.
pyttsx3>=2.90

# Optional: the legacy direct-mic listener (not used by the Gradio app).
# Uncomment if you want voice_cli to listen on a system microphone.
# SpeechRecognition>=3.10
# pyaudio>=0.2.11
